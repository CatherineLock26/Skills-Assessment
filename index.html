-- Use this after creating your first admin in Authentication > Users. -- Replace YOUR_ADMIN_EMAIL with the email address used for the admin Auth user. insert into public.profiles ( id, username, display_name, group_course, role, status, must_reset_password ) select id, lower(split_part(email, '@', 1)), coalesce(raw_user_meta_data ->> 'display_name', email), 'Admin', 'admin', 'active', false from auth.users where lower(email) = lower('YOUR_ADMIN_EMAIL') on conflict (id) do update set role = 'admin', status = 'active', must_reset_password = false;
